abi <abi/3.0>,

#include <tunables/global>

profile lva-supervisor flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/golang>

  # Core communication backplanes
  dbus,

  # Low-latency networking for local voice streaming and mDNS
  network unix stream,
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,

  # Container tracking signal mechanics
  signal (send) set=(kill,term,int,hup,cont),

  # Root privileges needed to manage network interfaces and file permissions
  capability net_admin,
  capability net_bind_service,
  capability dac_read_search,
  capability dac_override,
  capability chown,

  # Allowed execution binaries on the host/container
  /bin/** ix,
  /usr/bin/** ix,
  /bin/udevadm Ux,
  /sbin/udevd Ux,
  /usr/bin/lva ix,
  /usr/bin/git cx,
  
  # Process layout tracking permissions
  /etc/s6/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,

  # Prevent profile exploits from modifying vital kernel nodes
  deny /proc/** wl,
  deny /sys/** wl,

  # Global file read accessibility rules
  / r,
  /** r,
  
  # Dynamic variable paths required for your voice assistant 
  /tmp/** rwk,
  /etc/lva/** rw,
  /run/lva/{,**} rwk,
  /dev/tty rw,
  /etc/resolv.conf rw,
  /run/docker.sock rw,

  # Core dynamic engine shared library lookups
  /usr/local/lib/** mr,

  # Locked sub-profile for git pull updates
  profile /usr/bin/git flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>

    network,
    deny network raw,

    signal (receive) set=(term),

    /bin/busybox ix,
    /usr/bin/git mrix,
    /usr/libexec/git-core/* ix,

    # Guardrails to prevent git tasks from reading voice configuration tokens
    deny /etc/lva rw,

    /** r,
    /lib/* mr,
    /etc/lva/addons/** lrw,
    /etc/lva/tmp/** lrw,
    /usr/local/lib/** mr,

    capability dac_override,
  }
}
